# SDL.h renames main() to SDL_main() unless SDL_MAIN_HANDLED is defined.
# Without it, linking fails with an unresolved main/SDL_main.
# With it, call SDL_SetMainReady() before SDL_Init().
add_definitions(-DSDL_MAIN_HANDLED)

add_executable(
	Nillium
	"main.cpp"
	"app/app.h"
	"app/app.cpp"
	"assets/graphic.h"
	"assets/graphic.cpp"
	"assets/palette.h"
	"assets/palette.cpp"
	"assets/wad.h"
	"assets/wad.cpp"
	"platform/keys.h"
	"platform/video.h"
	"platform/video.cpp"
	"presentation/drawframe.h"
	"presentation/drawframe.cpp"
	"presentation/graphicstore.h"
	"presentation/graphicstore.cpp"
	"presentation/patchdrawer.h"
	"presentation/patchdrawer.cpp"
	"presentation/shaderprogram.h"
	"presentation/shaderprogram.cpp"
	"presentation/texture.h"
	"presentation/texture.cpp"
	"presentation/vertexarray.h"
	"presentation/vertexarray.cpp"
)

# BINARY_DIR: headers generated by configure_file() into the build tree.
# SOURCE_DIR: lets includes be written relative to src/.
target_include_directories(Nillium PRIVATE
	${CMAKE_CURRENT_BINARY_DIR}
	${CMAKE_CURRENT_SOURCE_DIR}
)

target_link_libraries(Nillium PRIVATE glad glfw)
